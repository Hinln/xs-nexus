cmake_minimum_required(VERSION 3.20)
project(xsnet_abi C)

add_library(xsnet_abi STATIC src/abi.c src/session.c)
target_include_directories(xsnet_abi PUBLIC include)
target_compile_features(xsnet_abi PUBLIC c_std_11)

if(MSVC)
    target_compile_options(xsnet_abi PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(xsnet_abi PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(xsnet_abi_test tests/abi_test.c)
    target_link_libraries(xsnet_abi_test PRIVATE xsnet_abi)
    if(MSVC)
        target_compile_options(xsnet_abi_test PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(xsnet_abi_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME xsnet_abi_validation COMMAND xsnet_abi_test)

    add_executable(xsnet_session_test tests/session_test.c)
    target_link_libraries(xsnet_session_test PRIVATE xsnet_abi)
    if(MSVC)
        target_compile_options(xsnet_session_test PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(xsnet_session_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME xsnet_session_validation COMMAND xsnet_session_test)
endif()
